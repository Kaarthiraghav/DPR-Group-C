Standardize all missing values to true NAs

# Standardize various "Unknown" strings to actual NA values
# We apply this only to character columns to avoid type errors with numeric data
cleaned_data <- cleaned_data %>% 
  mutate(across(where(is.character), ~na_if(., "Unknown"))) %>%
  mutate(across(where(is.character), ~na_if(., "NA"))) %>%
  mutate(across(where(is.character), ~na_if(., "[Not Available]")))

Clean and validate primary features

cleaned_features <- cleaned_data %>% 
  # --- DEMOGRAPHICS (3 Columns) ---
  mutate(current_age = as.numeric(current_age)) %>% 
  filter(current_age > 0 & current_age < 110) %>% 
  mutate(sex = as.factor(sex)) %>% 
  mutate(race = as.factor(race)) %>% 

  # --- CLINICAL PATHOLOGY (4 Columns) ---
  mutate(cancer_type = as.factor(cancer_type)) %>% 
  mutate(cancer_type_detailed = as.factor(cancer_type_detailed)) %>% 
  mutate(primary_tumor_site = as.factor(primary_tumor_site)) %>% 
  
  # Simplify Stage
  mutate(stage_simplified = case_when(
    grepl("Stage 4|Metastatic|Distant", stage_highest_recorded) ~ "Advanced",
    grepl("Stage 1-3|Localized|Regional", stage_highest_recorded) ~ "Early/Regional",
    TRUE ~ NA_character_ 
  )) %>% 
  mutate(stage_simplified = as.factor(stage_simplified)) %>%

  # --- GENOMIC & MOLECULAR (5 Columns) ---
  # These are crucial for predicting response in sequencing data
  mutate(fraction_genome_altered = as.numeric(fraction_genome_altered)) %>% 
  mutate(mutation_count = as.numeric(mutation_count)) %>% 
  mutate(tmb_nonsynonymous = as.numeric(tmb_nonsynonymous)) %>% 
  mutate(tumor_purity = as.numeric(tumor_purity)) %>% 
  mutate(msi_type = as.factor(msi_type)) %>% # Microsatellite instability impacts treatment

  # --- NLP / HISTORY (2 Columns) ---
  mutate(smoking_history_nlp = as.factor(smoking_history_nlp)) %>% 
  
  # --- TARGET VARIABLE ---
  mutate(response = if_else(overall_survival_status == "0:LIVING", "Responded", "Non-Responded")) %>% 
  mutate(response = as.factor(response))

Final selection of clean features for your analysis

final_analysis_df <- cleaned_features %>%
  select(
    # Demographics
    current_age, sex, race,
    # Clinical
    cancer_type, cancer_type_detailed, primary_tumor_site, stage_simplified,
    # Genomics (The "Data Programming" part of the module often looks for these numeric fields)
    fraction_genome_altered, mutation_count, tmb_nonsynonymous, tumor_purity, msi_type,
    # History
    smoking_history_nlp,
    # Target
    response
  ) %>%
  drop_na()
